# ============================================================================
# GADAVIRAL app at the site root  (add to /public_html/.htaccess)
# ============================================================================
# The site root is WordPress (index.php) whose front page renders empty —
# that is the white page. This block intercepts requests BEFORE index.php
# runs, so no WordPress file needs to change.
#
# HOW TO APPLY (cPanel → File Manager):
#   1. Settings (top-right) → tick "Show Hidden Files (dotfiles)" → Save.
#   2. In /public_html/ find .htaccess. If you don't see it, it is hidden —
#      do NOT skip this step. Download a backup copy first.
#   3. If .htaccess exists (it normally does — WordPress rewrites live there):
#      Edit it and paste EVERYTHING between the dashed lines below at the VERY
#      TOP, above the "# BEGIN WordPress" line. Save.
#      If .htaccess does NOT exist: create it with exactly the block below
#      (WordPress will append its own # BEGIN WordPress section on the next
#      permalink save — no conflict).
#
# Effect: the React app serves directly on https://www.gadaviral.com/ (no URL
# redirect) and on every SPA route below, while ALL WordPress paths
# (wp-admin, wp-login.php, wp-json, wp-content, wp-includes, wp-cron.php,
# xmlrpc.php, ...) keep working exactly as before.
#
# Use EITHER this block (clean root URL) OR deploy/root-index.html (redirect
# to /app/) — no need for both.
# ============================================================================

<IfModule mod_rewrite.c>
  RewriteEngine On

  # 1. Homepage → the app (internal rewrite; the URL stays the clean root)
  RewriteRule ^/?$ /app/index.html [L]

  # 2. App paths + SPA deep links → the app. The conditions let real files
  #    (the app's own /app/assets/*.js, CSS, icons, ...) pass through
  #    untouched; every wp-* path and anything unknown falls through to
  #    WordPress. The app/? alternative also makes /app/... deep links and
  #    refreshes work without relying on the /app/.htaccess fallback.
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteRule ^(app/?|login|register|verify|verified|reset-password|auth/google/complete|feed|explore|post(/.*)?|u(/.*)?|notifications|messages|groups(/.*)?|events|businesses|settings|admin)(/.*)?$ /app/index.html [L]
</IfModule>
