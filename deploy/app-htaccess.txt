# GADAVIRAL SPA fallback .htaccess — TWO variants, use the one for the target:
#
#   PRODUCTION  → place at /public_html/app/.htaccess        (block A, unchanged)
#   STAGING     → place at /public_html/staging/app/.htaccess (block B)
#
# Ensures React deep links like /feed, /u/<username> resolve to index.html.
# Never cache the SPA shell (asset filenames are hashed instead).

# ── Block A: PRODUCTION (/app/) ─────────────────────────────────────────────
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteBase /app/
  RewriteRule ^index\.html$ - [L]
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteRule . /app/index.html [L]
</IfModule>
<IfModule mod_headers.c>
  <FilesMatch "index\.html$">
    Header set Cache-Control "no-cache, must-revalidate"
  </FilesMatch>
</IfModule>
